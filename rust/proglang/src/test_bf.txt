loop {
    var instr;
    instr = prog[pc];
    if instr == Char::Plus {
        mem[mc]=(mem[mc]+1)&255;
    } else if instr == Char::Minus {
        mem[mc]=(mem[mc]-1)&255;
    } else if instr == Char::Greater {
        mc=mc+1;
    } else if instr == Char::Less {
        mc=mc-1;
    } else if instr == Char::Comma {
        // mem[mc]=read_char();
    } else if instr == Char::Period {
        // print_char(mem[mc]);
        print(mem[mc]);
    } else if instr == Char::BracketOpen {
        if mem[mc]==0 {
            var stack;
            stack=0;
            while stack!=0 || prog[pc] != Char::BracketClose {
                if prog[pc]==Char::BracketOpen  stack=stack+1;
                if prog[pc]==Char::BracketClose stack=stack-1;
                pc=pc+1;
            }
            // ends at bracket; +1 later skips the bracket again
        }
    } else if instr == Char::BracketClose {
        if mem[mc]!=0 {
            var stack;
            stack=0;
            while stack!=0 || prog[pc] != Char::BracketOpen {
                if prog[pc]==Char::BracketOpen  stack=stack-1;
                if prog[pc]==Char::BracketClose stack=stack+1;
                pc=pc-1;
            }
            // ends at bracket; +1 later skips the bracket again
        }
    }
    pc=pc+1;
}
